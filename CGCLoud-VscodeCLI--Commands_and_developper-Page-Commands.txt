(1) All your sample accounts - List

(2) All Retail Stores with their linked Account

SELECT Id, Name, AccountId, Account.Name, StoreType FROM RetailStore


(3). Retail Stores for a specific customer

SELECT Id, Name, StoreType FROM RetailStore WHERE Account.Name = 'SriLakshmi Supermarket'

(4) All sample products

SELECT Id, Name, ProductCode, Family, IsActive FROM Product2 WHERE Family = 'Consumer Goods'


(5) Products with their price book entries

SELECT Product2.Name, UnitPrice, IsActive FROM PricebookEntry WHERE Pricebook2.IsStandard = true


(6) Store Products — which products are sold at which stores

SELECT RetailStoreId, RetailStore.Name, ProductId, Product.Name, StartDate, EndDate FROM StoreProduct

(7) Count of products per store

SELECT RetailStore.Name, COUNT(Id) FROM StoreProduct GROUP BY RetailStore.Name
